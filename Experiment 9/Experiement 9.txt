#include <stdio.h>

int main() {
    int n, cost[10][10], visited[10] = {0};
    int i, j, min, u, v, total = 0, edges = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter adjacency matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            if (cost[i][j] == 0)
                cost[i][j] = 999;
        }

    visited[0] = 1;

    while (edges < n - 1) {
        min = 999;

        for (i = 0; i < n; i++)
            if (visited[i])
                for (j = 0; j < n; j++)
                    if (!visited[j] && cost[i][j] < min) {
                        min = cost[i][j];
                        u = i;
                        v = j;
                    }

        if (min == 999) {
            printf("MST not possible\n");
            return 0;
        }

        printf("%d - %d : %d\n", u + 1, v + 1, min);
        visited[v] = 1;
        total += min;
        edges++;
    }

    printf("Minimum cost = %d\n", total);

    return 0;
}
